<script>
const startBtn = document.getElementById("start");
const stopBtn = document.getElementById("stop");
const resetBtn = document.getElementById("reset");
const terminal = document.getElementById("terminal");
const bar = document.getElementById("bar");
const percent = document.getElementById("percent");
const statusText = document.getElementById("status");

const events = [
  ["تهيئة بيئة المحاكاة المحلية", "line"],
  ["بدء فحص تجريبي للواجهة", "line"],
  ["عرض تنبيه أمني افتراضي", "warning"],
  ["تحليل بيانات توضيحية غير حقيقية", "line"],
  ["محاكاة اكتشاف حدث أمني", "warning"],
  ["لا يوجد اتصال بخادم خارجي", "line"],
  ["لم يتم الوصول إلى ملفات النظام", "line"],
  ["اكتملت المحاكاة بنجاح", "line"]
];

let progress = 0;
let timer = null;
let running = false;

function addLine(message, type = "line") {
  const line = document.createElement("div");
  line.className = type;
  line.textContent = "[" + new Date().toLocaleTimeString("ar") +
    "] " + message;
  terminal.appendChild(line);
  terminal.scrollTop = terminal.scrollHeight;
}

function finish() {
  clearInterval(timer);
  timer = null;
  running = false;
  progress = 100;
  bar.style.width = "100%";
  percent.textContent = "100%";
  statusText.textContent = "اكتملت المحاكاة";
  addLine("النتيجة: لم يحدث أي اختراق حقيقي.", "line");
  startBtn.disabled = false;
  stopBtn.disabled = true;
}

function tick() {
  if (!running) return;

  progress = Math.min(progress + Math.floor(Math.random() * 8) + 3, 100);
  bar.style.width = progress + "%";
  percent.textContent = progress + "%";
  statusText.textContent = "جارٍ تنفيذ الفحص التجريبي...";

  const index = Math.min(
    Math.floor(progress / 100 * events.length),
    events.length - 1
  );

  if (progress >= 100) {
    addLine(events[events.length - 1][0], events[events.length - 1][1]);
    finish();
    return;
  }

  if (progress === 3 || progress % 15 < 8) {
    addLine(events[index][0], events[index][1]);
  }
}

startBtn.addEventListener("click", () => {
  if (running) return;

  if (progress >= 100) {
    progress = 0;
    bar.style.width = "0%";
    percent.textContent = "0%";
  }

  running = true;
  startBtn.disabled = true;
  stopBtn.disabled = false;
  addLine("بدأت المحاكاة المحلية.", "warning");
  tick();
  timer = setInterval(tick, 650);
});

stopBtn.addEventListener("click", () => {
  if (!running) return;
  clearInterval(timer);
  timer = null;
  running = false;
  statusText.textContent = "تم إيقاف المحاكاة";
  addLine("أوقف المستخدم المحاكاة.", "info");
  startBtn.disabled = false;
  stopBtn.disabled = true;
});

resetBtn.addEventListener("click", () => {
  clearInterval(timer);
  timer = null;
  running = false;
  progress = 0;
  bar.style.width = "0%";
  percent.textContent = "0%";
  statusText.textContent = "في انتظار بدء الفحص التجريبي";
  terminal.replaceChildren();
  addLine("تمت إعادة ضبط المحاكاة.", "line");
  startBtn.disabled = false;
  stopBtn.disabled = true;
});
</script>
